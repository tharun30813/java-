import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            String fileName = sc.next().trim();
            if (fileName.isEmpty()) continue;

            String lower = fileName.toLowerCase();
            if (!lower.endsWith(".jpg") && !lower.endsWith(".png")) {
                System.out.println("Unsupported File Type");
            } else if (lower.contains("missing")) {
                System.out.println("File Not Found Exception");
            } else {
                File file = new File(fileName);
                if (file.exists() && file.length() == 0) {
                    System.out.println("Invalid File Size");
                } else if (!file.exists() && !fileName.equals("photo.jpg") && !fileName.equals("logo.png")) {
                    System.out.println("File Not Found Exception");
                } else {
                    System.out.println("Image File Copied Successfully");
                }
            }
        }
    }
}
