import java.util.*;
import java.util.regex.*;

interface Payment {
    String processPayment();
}

class PaymentProcessor implements Payment {
    private String mode;
    public PaymentProcessor(String mode) {
        this.mode = mode;
    }
    public String processPayment() {
        return "Payment Successful";
    }
}

abstract class Item {
    protected String id;
    protected String name;
    protected double price;
    public Item(String id, String name, double price) {
        this.id = id;
        this.name = name;
        this.price = price;
    }
    public abstract double calculateFinalPrice(double discount);
}

class Product extends Item {
    public Product(String id, String name, double price) {
        super(id, name, price);
    }

    public double calculateFinalPrice() {
        return this.price;
    }

    @Override
    public double calculateFinalPrice(double discount) {
        return this.price - (this.price * discount / 100.0);
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNextLine()) {
            String line = sc.nextLine().trim();
            if (line.isEmpty()) continue;

            String pId = "";
            String pName = "";
            double price = -1;
            double discount = 0;
            String paymentMode = "";

            Matcher mId = Pattern.compile("Product ID:\\s*([^,\n]+)").matcher(line);
            if (mId.find()) pId = mId.group(1).trim();

            Matcher mName = Pattern.compile("Product Name:\\s*([^,\n]+)").matcher(line);
            if (mName.find()) pName = mName.group(1).trim();

            Matcher mPrice = Pattern.compile("Price:\\s*[₹]?([0-9,]+(\\.[0-9]+)?)").matcher(line);
            if (mPrice.find()) price = Double.parseDouble(mPrice.group(1).replace(",", ""));

            Matcher mDisc = Pattern.compile("Discount:\\s*([0-9,]+(\\.[0-9]+)?)\\s*%?").matcher(line);
            if (mDisc.find()) discount = Double.parseDouble(mDisc.group(1).replace(",", ""));

            Matcher mMode = Pattern.compile("Payment Mode:\\s*([^,\n]+)").matcher(line);
            if (mMode.find()) paymentMode = mMode.group(1).trim();

            if (price <= 0) {
                System.out.println("Invalid Product Price");
            } else {
                Product p = new Product(pId, pName, price);
                double finalPrice = p.calculateFinalPrice(discount);
                Payment pm = new PaymentProcessor(paymentMode);
                System.out.printf(Locale.US, "Final Price = ₹%,d, %s\n", (long) finalPrice, pm.processPayment());
            }
        }
    }
}
