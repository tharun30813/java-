import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) return;
        String line = sc.nextLine().trim();
        if (line.isEmpty()) return;

        String[] parts = line.split(",");
        if (parts.length < 3) return;

        int empId;
        try {
            empId = Integer.parseInt(parts[0].trim());
        } catch (NumberFormatException e) {
            return;
        }

        String empName = parts[1].trim();

        StringBuilder attBuilder = new StringBuilder();
        for (int i = 2; i < parts.length; i++) {
            attBuilder.append(parts[i]).append(" ");
        }
        while (sc.hasNext()) {
            attBuilder.append(sc.next()).append(" ");
        }

        Scanner attSc = new Scanner(attBuilder.toString());
        List<Integer> attList = new ArrayList<>();
        while (attSc.hasNextInt()) {
            attList.add(attSc.nextInt());
        }

        if (attList.size() < 7) {
            System.out.println("Invalid Attendance Input");
            return;
        }

        int presentCount = 0;
        for (int i = 0; i < 7; i++) {
            int val = attList.get(i);
            if (val != 0 && val != 1) {
                System.out.println("Invalid Attendance Input");
                return;
            }
            if (val == 1) presentCount++;
        }

        int absentDays = 7 - presentCount;
        double percentage = (presentCount / 7.0) * 100.0;
        String eligibility = (percentage >= 90.0) ? "Eligible" : "Not Eligible";

        System.out.printf(Locale.US, "Attendance = %.2f%%, Absent Days = %d, %s\n", percentage, absentDays, eligibility);
    }
}
