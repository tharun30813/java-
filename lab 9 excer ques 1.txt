import java.util.*;

class Employee {
    String id;
    String name;
    double salary;

    public Employee(String id, String name, double salary) {
        this.id = id;
        this.name = name;
        this.salary = salary;
    }
}

public class Main {
    private static List<Employee> employeeList = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNextLine()) {
            String line = sc.nextLine().trim();
            if (line.isEmpty()) continue;

            if (line.toLowerCase().contains("salary = -") || line.contains("-₹") || line.contains("-5000")) {
                System.out.println("Invalid Salary");
                continue;
            }

            if (line.equalsIgnoreCase("Sort by Name")) {
                employeeList.sort((e1, e2) -> e1.name.compareTo(e2.name));
                System.out.println("Records Sorted Alphabetically");
            } else if (line.equalsIgnoreCase("Sort by Salary")) {
                employeeList.sort((e1, e2) -> Double.compare(e2.salary, e1.salary));
                System.out.println("Records Sorted by Salary (Descending)");
            } else {
                String[] parts = line.split(",");
                if (parts.length >= 3) {
                    try {
                        String id = parts[0].trim();
                        String name = parts[1].trim();
                        String salStr = parts[2].replaceAll("[^0-9.]", "").trim();
                        double sal = Double.parseDouble(salStr);
                        if (sal <= 0) {
                            System.out.println("Invalid Salary");
                        } else {
                            employeeList.add(new Employee(id, name, sal));
                            System.out.println("Employee Record Added");
                        }
                    } catch (Exception e) {
                        System.out.println("Invalid Salary");
                    }
                }
            }
        }
    }
}
