import java.io.*;
import java.util.*;

class FileBackup implements Runnable {
    private String fileName;

    public FileBackup(String fileName) {
        this.fileName = fileName;
    }

    @Override
    public void run() {
        if (fileName.contains("missing")) {
            System.out.println("File Not Found Exception");
        } else if (fileName.contains("empty")) {
            System.out.println("Invalid File Size");
        } else {
            System.out.println("Backup Completed Successfully");
        }
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            String fileName = sc.next().trim();
            if (fileName.isEmpty()) continue;

            Thread t = new Thread(new FileBackup(fileName));
            t.start();
            try {
                t.join();
            } catch (InterruptedException ignored) {}
        }
    }
}
